from typing import Literal

import numpy as np

from polygeo import (
    BoundaryRegular,
    BoundaryUnknown,
    CochainSpace,
    CochainSubspace,
    Complex,
    ConnectivityUnknown,
    Geometry,
    OrientationUnknown,
    Simplicial,
    extend_zero,
    restrict,
    hodge_star,
    topological_boundary,
)

type Raw = Complex[
    BoundaryUnknown,
    OrientationUnknown,
    ConnectivityUnknown,
    Simplicial,
]
type Regular = Complex[
    BoundaryUnknown,
    OrientationUnknown,
    ConnectivityUnknown,
    BoundaryRegular,
]

raw: Raw = Complex.from_maximal_simplices(
    np.array([[0, 1, 2]], dtype=np.int64)
)
topological_boundary(raw)
regular: Regular = raw.boundary_regular()
regular.boundary_regular()

C0 = regular.cochain_space(0)
C1 = regular.cochain_space(1)
edge = CochainSubspace(C1, np.array([0], dtype=np.int64))
restrict(C0, edge)
extend_zero(edge, C0)
raw_parent = raw.cochain_space(1)
restrict(raw_parent, edge)
extend_zero(edge, raw_parent)
geometry = Geometry.from_positions(raw, np.eye(3, dtype=np.float64))
dual = hodge_star(geometry, raw.cochain_space(1)).target
CochainSubspace(dual, np.array([0], dtype=np.int64))


def consume_vertex_subspace(
    value: CochainSubspace[CochainSpace[Regular, Literal[0]]],
) -> None:
    pass


consume_vertex_subspace(edge)
